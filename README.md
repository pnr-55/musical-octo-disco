

import streamlit as st

st.title("ระบบคัดกรองและแนะนำการบำบัดข้อเข่า/รองช้ำ อัจฉริยะ")
st.write("ทีม Vitamin C - โรงเรียนพระนารายณ์")

# 1. ส่วนลงทะเบียนและเลือกโรงพยาบาลเครือข่าย
st.header("1. ข้อมูลผู้ป่วย")
name = st.text_input("ชื่อ-นามสกุล ผู้ป่วย")
hospital = st.selectbox("ส่งต่อข้อมูลไปยัง", ["รพ.สต. ในพื้นที่", "โรงพยาบาลพัฒนานิคม", "โรงพยาบาลพระนารายณ์มหาราช"])

# 2. ฟังก์ชันประเมินอาการ (สามารถใส่โค้ด MediaPipe วิเคราะห์เข่าเดิมของหนูตรงนี้ได้)
st.header("2. การประเมินอาการปวด")
symptom = st.radio("จุดที่มีอาการปวดหลัก", ["ข้อเข่า (ขาโก่ง/ขาฉิ่ง)", "ส้นเท้า/ฝ่าเท้า (รองช้ำ)"])

# 3. ส่วนอัปเดต: ระบบแนะนำการใช้แผ่นประคบดินสอพอง-สมุนไพร อัตโนมัติ
st.header("3. ผลการวิเคราะห์และคำแนะนำการบำบัด")
if st.button("ประมวลผลข้อมูล"):
st.success(#คำแนะนำที่เชื่อมโยงกับผลิตภัณฑ์ใหม่)
if symptom == "ส้นเท้า/ฝ่าเท้า (รองช้ำ)":
st.write(f"**คำแนะนำสำหรับคุณ {name}:**")
st.write("- **สูตรประคบร้อน:** นำแผ่นประคบดินสอพองเข้าไมโครเวฟไฟกลาง 1-2 นาที แล้วนำมาเหยียบคึงเบาๆ 15 นาที เพื่อคลายพังผืด")
st.write("- **สูตรประคบเย็น:** หากมีอาการปวดบวมเฉียบพลัน ให้แช่แผ่นประคบในช่องฟรีซ 1 ชั่วโมงก่อนนำมาเหยียบ")
else:
st.write("แนะนำโปรแกรมยืดเหยียดข้อเข่า และใช้สายรัดประคบร้อนสมุนไพรบริเวณรอบข้อเข่า")

st.info(f"ระบบได้ทำการส่งข้อมูลการประเมินเบื้องต้นไปยัง {hospital} เรียบร้อยแล้ว")
